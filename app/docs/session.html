Sessions

LilaPHP supports normal and encrypted sessions. Encrypted sessions use AES-256-GCM internally for secure storage. 

// Basic session usage
use Core\Session;
use Core\App;
use Core\GET;
use Core\POST;

$app=new App();

//global with App()
$app->setSession("user_id", 1);
$user = $app->getSession("user_id");
$app->removeSession("user_id");

#[GET]
function loginPost($req, Response $res,Session $session)
{
    $session::setSession("user_id", 1);
    $user=$session::getSession("user_id");
    $session::removeSession("user_id");
}
#[POST]
function loginPost($req, Response $res,Session $session)
{
    $session::setSession("user_id", 1);
    $user=$session::getSession("user_id");
    $session::removeSession("user_id");
}

// Encrypted session (recommended for authentication)
use Core\App;
use Core\GET;
use Core\POST;
use Core\Session;

$app=new App();
//global with App()
$user = $app->getSession(
    key: "auth",
    default: null,
    decrypt: true
);
$app->setSession(
    key: "auth",
    value: $user,
    encrypt: true
);

When encrypt: true is used, session data is encrypted before being stored. Use decrypt: true when retrieving secure session values.

Basic Login Example (Secure Session) 

// LilaPHP/routes/login.php
include_once __DIR__ . "/../index.php";

use Core\{App, BaseModel, Field, CSRF, POST, Validate, Response, Session, Database};

$app = new App();

class LoginModel extends BaseModel
{
    #[Field(required: true, format: "email")]
    public string $email;

    #[Field(required: true, min_length: 6)]
    public string $password;
}

#[POST] #[CSRF] #[Validate(LoginModel::class)]
function loginPost(array $req, Response $res, Session $session, Database $db)
{
    $pdo = $db->getConnection();
    $email = trim($req["email"] ?? "");
    $password = trim($req["password"] ?? "");

    $stmt = $pdo->prepare("SELECT id, name, password FROM users WHERE email = ? AND is_active = 1");
    $stmt->execute([$email]);
    $user = $stmt->fetchObject();

    if ($user && password_verify($password, $user->password)) {
        unset($user->password);
        session_regenerate_id(true);

        $session::set(key: "auth", value: $user, encrypt: true);

        return $res->jsonResponse(["success" => true, "user" => $user]);
    }

    return $res->jsonResponse(["success" => false, "message" => "Invalid credentials"], status: 401);
}

$app->add('loginPost');
$app->run();
