LilaPHP

Routing

Zero-config routing system with method injection and middleware support.

GET, POST, PUT, DELETE, PATCH, OPTIONS... 

// LilaPHP/routes/index.php
include_once __DIR__ . "/../index.php";

use Core\{App, Response};

$app = new App();
$app->post(
    callback: function(array $req, Response $res) {
        return $res->jsonResponse(["success" => true]);
    },
    csrf: true,
    middlewares: [fn($req, $res) => new LoginModel(data: $req, lang: $lang)]
);
$app->run();

Routing Attributes

New

Use PHP 8 Attributes to define HTTP methods, CSRF protection, and middlewares directly on your handlers.

Available Attributes

  • #[GET], #[POST]
  • #[PUT], #[DELETE]
  • #[CSRF] - Enable CSRF validation
Example Usage 

// LilaPHP/routes/index.php
include_once __DIR__ . "/../index.php";
use Core\GET;
use Core\POST;
use Core\CSRF; 
use Core\App;
use Core\Response;

$app = new App();

// Register using attributes for functions or methods

#[GET]
function login(array $req, Response $res)
{
    return $res->render("login");
}
$app->add(callback: 'login');

#[POST]
#[CSRF]
function loginPost(array $req, Response $res)
{
    return $res->jsonResponse(["success" => true]);
}
$app->add(callback: 'loginPost', middlewares: [fn(array $req) => new LoginModel(data: $req, lang: $lang)]);

//With class
class Example{
    #[GET]
    static function GET(array $req, Response $res){
        return $res->render("login");
    }
}
$app->add(callback:  [Example::class,'GET'] );

Auto-wiring Dependency Injection

New

LilaPHP automatically resolves your route handler dependencies using Reflection. This means you no longer need global $app; or to pass specific arguments manually. 

use Core\Response;
use Core\Session;
use Core\Config;
use Core\Database;
use Core\GET;

#[GET]
function showDashboard(array $req, Response $res, Session $session) {
    // The framework detects that you need the Request array, Response object, and Session.
    // It automatically injects them without any extra work.
    
    if (!$session::has('user')) {
        return $res->redirect('/login');
    }
    
    return $res->render('dashboard');
}

#[GET]
function showDashboardWithConfigAndDB(array $req, Response $res, Config $config, Database $db) {
    // $db is automatically instantiated using configuration from .env by default
    $pdo = $db->getConnection();
    $insert = $pdo->prepare("INSERT INTO users (name, email, password) VALUES (?, ?, ?)");
    $random = rand(1, 100);
    $email = "Jhon{$random}@email.com";
    $insert->execute(['John Doe', $email, 'password']);
    $query = $db->query("SELECT * FROM users");
    $users = $query->fetchAll(PDO::FETCH_ASSOC); 
    
    
    $debug = $config::$DEBUG;
    $urlProject = $config::$URL_PROJECT;
    
    return $res->render('dashboard');
}

Order independent and Custom Classes

The order of parameters does not matter. The framework can also auto-instantiate any custom class for you as long as it has a zero-argument constructor. 

use Utils\EmailService;

#[POST]
function sendEmail(Response $res, EmailService $mailer, array $req) {
    // $res is injected successfully regardless of being the first parameter!
    // $mailer is automatically instantiated via new EmailService()
    
    $mailer->send($req['to'], "Welcome!");
    return $res->jsonResponse(['sent' => true]);
}
