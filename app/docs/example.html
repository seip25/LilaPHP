Learn how all LilaPHP components work together by building a Newsletter Signup system with asynchronous processing.

1 The Database Model

models/SubscriberModel.php
class SubscriberModel extends BaseModel {
    #[FieldDatabase(type: 'int', primaryKey: true, autoIncrement: true)]
    public int $id;

    #[Field(required: true, min_length: 2)]
    #[FieldDatabase(type: 'varchar', length: 100)]
    public string $name;

    #[Field(required: true, format: 'email')]
    #[FieldDatabase(type: 'varchar', length: 255, unique: true)]
    public string $email;
}

2 The Background Job

resources/templates/newsletter.twig
namespace Jobs;

class WelcomeEmailJob {
    public function handle(array $payload) {
        // Logic to send external email...
        Logger::info("Sent welcome to " . $payload['email']);
    }
}

3 The Endpoint Controller

routes/newsletter.php
#[POST] #[CSRF]
function handleSignup(array $req, Response $res) {
    // 1. Validates and saves
    $sub = new SubscriberModel(data: $req);
    $sub->save();

    // 2. Dispatched to background worker
    Queue::push(WelcomeEmailJob::class, $req);

    return $res->redirect('newsletter/');
}
